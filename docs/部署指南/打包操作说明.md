
# 重新构建并部署-修改后需要重新构建后端镜像并重启容器：
 
# 1. 进入后端代码目录
cd ~/xuhz/deep-ems0/zhurong-ems-system

# 2. 编译打包
mvn clean package -DskipTests

# 3. 复制新的 jar 包到后端目录
cp target/zhurong-ems-system-4.6.0.jar ~/xuhz/docker/zhurong-ems/backend/app.jar

# 4. 停止并删除旧容器
docker compose -f ~/xuhz/docker/docker-compose-54.yml down zhurong-ems-backend

# 5. 重新启动
docker compose -f ~/xuhz/docker/docker-compose-54.yml up -d zhurong-ems-backend

# 6. 等待启动并验证
sleep 30 && docker logs zhurong-ems-backend --tail 30

# 进入项目目录
cd ~/xuhz/deep-ems0
# 构建镜像（-t 指定标签名称，-f 指定 Dockerfile 路径）
docker build -t zhurong-ems-backend:4.6.0 -f zhurong-ems-admin/Dockerfile .

## docker build 默认会复用缓存层。当 COPY . . 那步命中缓存时，源码没有重新复制进去，使用的还是旧的。强制不使用缓存：
docker build -t zhurong-ems-backend:4.6.0 -f zhurong-ems-admin/Dockerfile . --no-cache
## 或者更精细地让 Maven 那步跳过缓存：
docker build -t zhurong-ems-backend:4.6.0 -f zhurong-ems-admin/Dockerfile . --build-arg CACHEBUST=$(date +%s)

【这个会重新编译、打包jar】

# 验证镜像是否生成
docker images | grep zhurong-ems-backend

# 停止并删除旧容器
docker compose -f docker-compose-54.yml rm -s -f zhurong-ems-backend

# 重新启动
docker compose -f docker-compose-54.yml up -d zhurong-ems-backend

# 等待启动（约60秒），查看日志确认启动成功
docker logs -f zhurong-ems-backend

如果日志中出现 Started cpEmsApplication in X.XX seconds 且没有 Ambiguous mapping 报错，再用健康检查接口验证：

curl http://localhost:1088/autoee-iot-ems/actuator/health

# 保存镜像
docker save -o /mnt/d/zhurong-ems-backend-4.6.0.tar zhurong-ems-backend:4.6.0
